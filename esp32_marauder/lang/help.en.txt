[terms]

[terms.eapol]

a = SAE: Simultaneous Auth. of Equals, WPA3.

[terms.pmkid]
a = PMKID: Pairwise Master Key Identifier.

[terms.bssid]
a = BSSID: the hardware address of an AP.

[terms.ssid]
a = SSID: the name of a network.

[terms.promiscuous]
a = Promiscuous: hears all traffic, not just ours.

[terms.deauth]
a = Deauth: a frame that forces a rejoin.

[terms.pcapng]
a = pcapng: capture file that keeps notes.
